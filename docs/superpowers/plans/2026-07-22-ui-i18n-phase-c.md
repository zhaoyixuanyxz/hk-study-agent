# Phase C UI i18n Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.  
> **Status:** Implemented 2026-07-22 (TDD); verify with `cd frontend && npm test` and backend bilingual/intel DTO tests.

**Goal:** Make the user-facing shell (auth, nav, chat/profile/plan/report/history, dynamic `copy.js` strings, intelligence card titles) follow the **current thread’s `locale`**, switching instantly without a full page reload.

**Architecture:** Add a tiny i18n module (`t` + `zh`/`en` catalogs + `applyUiLocale`). Mark static HTML with `data-i18n*`. On `loadThread` / locale pick / login, set `document.documentElement.lang` and re-apply strings. Intelligence `en` prefers original title/snippet from API (expose fields if missing).

**Tech Stack:** Vanilla ES modules (existing Vite frontend), `node:test`, no new i18n library.

**Spec:** `docs/superpowers/specs/2026-07-22-ui-i18n-phase-c-design.md`

- [ ] **Step 4: Implement `index.js`**

```js
// frontend/src/i18n/index.js
import { catalogs } from "./catalogs.js";
import { LS_PREFERRED_LOCALE } from "../api/client.js";

/** @type {"zh"|"en"} */
let current = "zh";

/** @param {unknown} value @returns {"zh"|"en"} */
export function normalizeUiLocale(value) {
  const v = String(value || "").trim().toLowerCase();
  if (v === "en" || v === "en-us" || v === "en_gb") return "en";
  return "zh";
}

export function getUiLocale() {
  return current;
}

/** @param {"zh"|"en"|string} locale */
export function setUiLocale(locale) {
  current = normalizeUiLocale(locale);
  return current;
}

/**
 * @param {string} key
 * @param {Record<string, string|number>} [vars]
 */
export function t(key, vars) {
  const table = catalogs[current] || catalogs.zh;
  let s = table[key] ?? catalogs.zh[key] ?? key;
  if (vars) {
    for (const [k, v] of Object.entries(vars)) {
      s = s.replaceAll(`{${k}}`, String(v));
    }
  }
  return s;
}

/**
 * Resolve UI locale from thread or preferred storage.
 * @param {{ locale?: string } | null | undefined} thread
 */
export function resolveUiLocaleFromThread(thread) {
  if (thread && (thread.locale === "en" || thread.locale === "zh")) {
    return normalizeUiLocale(thread.locale);
  }
  if (typeof localStorage !== "undefined") {
    return normalizeUiLocale(localStorage.getItem(LS_PREFERRED_LOCALE));
  }
  return "zh";
}

/** @param {"zh"|"en"|string} [locale] */
export function applyUiLocale(locale) {
  const loc = setUiLocale(locale ?? current);
  if (typeof document === "undefined") return loc;

  document.documentElement.lang = loc === "en" ? "en" : "zh-CN";
  const title = t("doc.title");
  if (title) document.title = title;

  document.querySelectorAll("[data-i18n]").forEach((node) => {
    const key = node.getAttribute("data-i18n");
    if (!key) return;
    // Preserve child icons: prefer updating a nested [data-i18n-text] if present
    const textHost = node.querySelector("[data-i18n-text]") || node;
    if (textHost.childElementCount && !node.querySelector("[data-i18n-text]")) {
      // e.g. <strong>/<small> pairs handled via separate keys on those elements
      textHost.textContent = t(key);
    } else if (node.querySelector("[data-i18n-text]")) {
      node.querySelector("[data-i18n-text]").textContent = t(key);
    } else {
      node.textContent = t(key);
    }
  });

  document.querySelectorAll("[data-i18n-placeholder]").forEach((node) => {
    const key = node.getAttribute("data-i18n-placeholder");
    if (key && "placeholder" in node) node.placeholder = t(key);
  });

  document.querySelectorAll("[data-i18n-aria]").forEach((node) => {
    const key = node.getAttribute("data-i18n-aria");
    if (key) node.setAttribute("aria-label", t(key));
  });

  // Nested strong/small nav labels
  document.querySelectorAll("[data-i18n-strong]").forEach((node) => {
    const key = node.getAttribute("data-i18n-strong");
    if (key) node.textContent = t(key);
  });
  document.querySelectorAll("[data-i18n-small]").forEach((node) => {
    const key = node.getAttribute("data-i18n-small");
    if (key) node.textContent = t(key);
  });

  return loc;
}
```

Refine `data-i18n` vs `data-i18n-strong`/`small` so nav icons are not wiped (see Task 2 HTML pattern).
